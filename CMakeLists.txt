cmake_minimum_required(VERSION 3.25)
project(resonance VERSION 1.0.0 LANGUAGES C CXX ASM)

option(BUILD_DOCS "Build documentation alongside the sources." OFF)
option(BUILD_DOCS_ONLY "Build documentation and nothing else." OFF)
option(INSTALL_DOCS "Install documentation." OFF)
option(ENABLE_PATCHES "Include patch-only code in the build." ON)
set(VIDEO_STANDARD
    "NTSC"
    CACHE STRING "Release to build, NTSC for North America or PAL for Europe. PAL includes every \
change of the European release.")
set_property(CACHE VIDEO_STANDARD PROPERTY STRINGS NTSC PAL)
string(TOUPPER "${VIDEO_STANDARD}" VIDEO_STANDARD)
if(NOT VIDEO_STANDARD MATCHES "^(NTSC|PAL)$")
  message(FATAL_ERROR "VIDEO_STANDARD must be NTSC or PAL.")
endif()
if(NOT BUILD_DOCS_ONLY)
  include(cmake/tools.cmake)
endif()
include(cmake/disc-region.cmake)

if(ENABLE_PATCHES)
  add_compile_definitions(ENABLE_PATCHES)
endif()
if(VIDEO_STANDARD STREQUAL "PAL")
  add_compile_definitions(VIDEO_STANDARD_PAL)
endif()

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

if(BUILD_DOCS OR BUILD_DOCS_ONLY)
  set(DOXYGEN_GENERATE_HTML YES)
  set(DOXYGEN_HTML_OUTPUT "${CMAKE_BINARY_DIR}/html")
  set(DOXYGEN_JAVADOC_AUTOBRIEF YES)
  set(DOXYGEN_MARKDOWN_ID_STYLE GITHUB)
  set(DOXYGEN_USE_MDFILE_AS_MAINPAGE "${CMAKE_SOURCE_DIR}/README.md")
  set(DOXYGEN_INTERACTIVE_SVG YES)
  # Message, MetScreen, and the most included headers have more than 50 nodes in their graphs.
  set(DOXYGEN_DOT_GRAPH_MAX_NODES 200)
  set(DOXYGEN_GENERATE_TAGFILE "${CMAKE_BINARY_DIR}/html/${CMAKE_PROJECT_NAME}.tags")
  set(DOXYGEN_EXTRACT_ALL YES)
  set(DOXYGEN_EXTRACT_STATIC YES)
  set(DOXYGEN_SORT_MEMBER_DOCS NO)
  set(DOXYGEN_ENABLE_PREPROCESSING YES)
  set(DOXYGEN_MACRO_EXPANSION YES)
  set(DOXYGEN_EXPAND_ONLY_PREDEF YES)
  set(DOXYGEN_PREDEFINED "_EE")
  # Every reconstructed routine records the address it was recovered from with this tag.
  set(DOXYGEN_VERBATIM_VARS DOXYGEN_ALIASES)
  set(DOXYGEN_ALIASES [[ghidraAddress="\par Ghidra address:^^"]])
  set(DOXYGEN_EXCLUDE_PATTERNS "*/node_modules/*" "*/build/*" "*/src/python/*")
  # Headers sit beside their implementations under src and tools, and only headers are documented.
  set(DOXYGEN_FILE_PATTERNS "*.h")
  find_package(Doxygen REQUIRED COMPONENTS dot)
  doxygen_add_docs(docs README.md src tools ALL)
  if(INSTALL_DOCS)
    install(
      DIRECTORY "${CMAKE_BINARY_DIR}/html"
      DESTINATION "${CMAKE_INSTALL_DATAROOTDIR}/doc/${CMAKE_PROJECT_NAME}-${CMAKE_PROJECT_VERSION}")
  endif()
endif()

# A docs-only configure must not require the cross toolchain. The sources are skipped entirely.
if(NOT BUILD_DOCS_ONLY)
  include(FeatureSummary)
  include(GNUInstallDirs)
  feature_summary(WHAT ALL FATAL_ON_MISSING_REQUIRED_PACKAGES)
  include(cmake/ieee-single-plugin.cmake)
  add_subdirectory(src)
  add_subdirectory(sce/ee)
  include(cmake/file-basenames.cmake)
  resonance_basename_file_macros("${CMAKE_SOURCE_DIR}/src" "${CMAKE_SOURCE_DIR}/sce/ee")
  include(cmake/ezmidi-irx.cmake)
  include(cmake/image.cmake)
endif()
